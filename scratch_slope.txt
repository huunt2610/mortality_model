old_age_slope = {}
for year in years_detail:
    sub = np.log(mx_total.loc[60:90, year])
    slope = np.polyfit(sub.index.values, sub.values, 1)[0]
    old_age_slope[year] = round(slope, 4)

old_age_slope = pd.Series(old_age_slope, name="độ dốc log-linear tuổi 60-90 (mỗi năm tuổi)").to_frame()
